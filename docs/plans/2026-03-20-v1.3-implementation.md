**Step 2: Implement RoleManager**

```dart
// lib/services/field_link/role_manager.dart
import 'package:red_grid_link/data/models/team_role.dart';

class RoleManager {
  final String localDeviceId;
  TeamRole _localRole = TeamRole.scout;
  final Map<String, TeamRole> _peerRoles = {};
  String? _callsign;
  final Map<String, String> _peerCallsigns = {};

  RoleManager({required this.localDeviceId});

  TeamRole get localRole => _localRole;
  String get callsign => _callsign ?? '';

  void setCallsign(String value) => _callsign = value;

  void initializeAsCreator() {
    _localRole = TeamRole.lead;
  }

  void initializeAsJoiner() {
    _localRole = TeamRole.scout;
  }

  TeamRole roleForPeer(String peerId) {
    return _peerRoles[peerId] ?? TeamRole.scout;
  }

  String callsignForPeer(String peerId) {
    return _peerCallsigns[peerId] ?? '';
  }

  bool get isLead => _localRole.isLead;

  bool assignRole(String peerId, TeamRole role, {String? customLabel}) {
    if (!isLead) return false;
    _peerRoles[peerId] = role;
    return true;
  }

  void promotePeerToLead(String peerId) {
    if (!isLead) return;
    _peerRoles[peerId] = TeamRole.lead;
    _localRole = TeamRole.scout;
  }

  void applyRemoteRoleChange(String targetId, TeamRole role, {required String fromLeader}) {
    if (targetId == localDeviceId) {
      _localRole = role;
    } else {
      _peerRoles[targetId] = role;
    }
  }

  void applyRemoteCallsign(String peerId, String callsign) {
    _peerCallsigns[peerId] = callsign;
  }

  Map<String, dynamic>? encodeRoleAssignment(String targetId, TeamRole role, {String? customLabel}) {
    if (!isLead) return null;
    return {
      'evt': 'role_assign',
      'target': targetId,
      'role': role.toShortString(),
      if (customLabel != null) 'crl': customLabel,
    };
  }

  Map<String, dynamic> encodeCallsignUpdate() {
    return {
      'evt': 'callsign_update',
      'cs': callsign,
    };
  }

  void handleControlEvent(Map<String, dynamic> data, String senderId) {
    final evt = data['evt'] as String?;
    switch (evt) {
      case 'role_assign':
        final target = data['target'] as String;
        final role = TeamRole.fromString(data['role'] as String);
        applyRemoteRoleChange(target, role, fromLeader: senderId);
        break;
      case 'callsign_update':
        final cs = data['cs'] as String? ?? '';
        applyRemoteCallsign(senderId, cs);
        break;
    }
  }

  void reset() {
    _localRole = TeamRole.scout;
    _peerRoles.clear();
    _callsign = null;
    _peerCallsigns.clear();
  }
}
```

// lib/services/field_link/boundary_manager.dart
import 'dart:math';
import 'package:red_grid_link/data/models/annotation.dart';

class BoundaryManager {
  Annotation? _boundary;
  final Map<String, bool> _wasInside = {};

  Annotation? get boundary => _boundary;
  bool get hasBoundary => _boundary != null;

  void setBoundary(Annotation boundary) {
    _boundary = boundary;
    _wasInside.clear();
    // Assume all peers start inside on boundary creation
  }

  void clearBoundary() {
    _boundary = null;
    _wasInside.clear();
  }

  bool isInsideBoundary(double lat, double lon) {
    if (_boundary == null) return true;
    return _pointInPolygon(lat, lon, _boundary!.points);
  }

  /// Returns true if peer just crossed OUT of the boundary (inside->outside transition).
  bool checkBoundaryCrossing(String peerId, double lat, double lon) {
    if (_boundary == null) return false;

    final isInside = _pointInPolygon(lat, lon, _boundary!.points);
    final wasInside = _wasInside[peerId] ?? true; // assume inside on first check
    _wasInside[peerId] = isInside;

    return wasInside && !isInside; // transition from inside to outside
  }

  /// Ray casting algorithm for point-in-polygon.
  bool _pointInPolygon(double lat, double lon, List<AnnotationPoint> polygon) {
    if (polygon.length < 3) return false;

    bool inside = false;
    int j = polygon.length - 1;

    for (int i = 0; i < polygon.length; i++) {
      final yi = polygon[i].lat;
      final xi = polygon[i].lon;
      final yj = polygon[j].lat;
      final xj = polygon[j].lon;

      if (((yi > lat) != (yj > lat)) &&
          (lon < (xj - xi) * (lat - yi) / (yj - yi) + xi)) {
        inside = !inside;
      }
      j = i;
    }

    return inside;
  }
}

// lib/services/voice/nato_phonetic.dart
class NatoPhonetic {
  static const _letters = {
    'A': 'ALFA', 'B': 'BRAVO', 'C': 'CHARLIE', 'D': 'DELTA',
    'E': 'ECHO', 'F': 'FOXTROT', 'G': 'GOLF', 'H': 'HOTEL',
    'I': 'INDIA', 'J': 'JULIETT', 'K': 'KILO', 'L': 'LIMA',
    'M': 'MIKE', 'N': 'NOVEMBER', 'O': 'OSCAR', 'P': 'PAPA',
    'Q': 'QUEBEC', 'R': 'ROMEO', 'S': 'SIERRA', 'T': 'TANGO',
    'U': 'UNIFORM', 'V': 'VICTOR', 'W': 'WHISKEY', 'X': 'X-RAY',
    'Y': 'YANKEE', 'Z': 'ZULU',
  };

  static const _digits = {
    '0': 'ZE-RO', '1': 'WUN', '2': 'TOO', '3': 'TREE',
    '4': 'FOW-ER', '5': 'FIFE', '6': 'SIX', '7': 'SEV-EN',
    '8': 'AIT', '9': 'NIN-ER',
  };

  static String letterToNato(String char) {
    return _letters[char.toUpperCase()] ?? char;
  }

  static String digitToNato(String char) {
    return _digits[char] ?? char;
  }

  static String charToNato(String char) {
    if (RegExp(r'[a-zA-Z]').hasMatch(char)) return letterToNato(char);
    if (RegExp(r'[0-9]').hasMatch(char)) return digitToNato(char);
    return char;
  }

  static String mgrsToNato(String mgrs) {
    return mgrs.split('').map(charToNato).join(', ');
  }

  static String buildCallout(String callsign, String mgrs) {
    final natoGrid = mgrsToNato(mgrs);
    return '$callsign, grid, $natoGrid';
  }
}

```dart
// lib/services/voice/voice_callout_service.dart
import 'dart:collection';
import 'package:flutter_tts/flutter_tts.dart';
import 'package:red_grid_link/services/voice/nato_phonetic.dart';

class VoiceCalloutService {
  final FlutterTts? _tts;
  final Queue<String> _queue = Queue();
  bool _isSpeaking = false;
  final Map<String, _LastCallout> _debounce = {};
  static const _debounceMs = 30000; // 30 seconds

  VoiceCalloutService({FlutterTts? tts}) : _tts = tts {
    _tts?.setCompletionHandler(() {
      _isSpeaking = false;
      _processQueue();
    });
    _tts?.setSpeechRate(0.45);
    _tts?.setPitch(1.0);
  }

  /// For testing without TTS engine.
  factory VoiceCalloutService.forTesting() => VoiceCalloutService(tts: null);

  bool tryEnqueue(String peerId, String callsign, String mgrs) {
    final gridPrefix = mgrs.length >= 6 ? mgrs.substring(0, 6) : mgrs;
    final now = DateTime.now().millisecondsSinceEpoch;
    final last = _debounce[peerId];

    if (last != null &&
        last.gridPrefix == gridPrefix &&
        (now - last.timestamp) < _debounceMs) {
      return false;
    }

    _debounce[peerId] = _LastCallout(gridPrefix: gridPrefix, timestamp: now);
    final text = NatoPhonetic.buildCallout(callsign, mgrs);
    _queue.add(text);
    _processQueue();
    return true;
  }

  void _processQueue() {
    if (_isSpeaking || _queue.isEmpty || _tts == null) return;
    _isSpeaking = true;
    final text = _queue.removeFirst();
    _tts.speak(text);
  }

  void stop() {
    _queue.clear();
    _tts?.stop();
    _isSpeaking = false;
  }

  void dispose() {
    stop();
  }
}

class _LastCallout {
  final String gridPrefix;
  final int timestamp;
  _LastCallout({required this.gridPrefix, required this.timestamp});
}

  }) {
    return {

      'session': {
        'id': sessionId,
        'name': sessionName,
        'mode': operationalMode,
        'createdAt': createdAt.toUtc().toIso8601String(),
      },
      'peers': peers.map((p) => p.toJson()).toList(),
      'markers': markers.map((m) => m.toJson()).toList(),
      'annotations': annotations
          .where((a) => a.type != AnnotationType.boundary)

          .toList(),
      'trackPoints': trackPoints.map((t) => t.toJson()).toList(),

    return const JsonEncoder.withIndent('  ').convert(json);
  }

  static String exportFilename(String sessionName, DateTime date) {
    final safeName = sessionName.toLowerCase().replaceAll(RegExp(r'[^a-z0-9]'), '_');
    final dateStr = '${date.year}-${date.month.toString().padLeft(2, '0')}-${date.day.toString().padLeft(2, '0')}';
    return 'redgridlink_session_${safeName}_$dateStr.json';
  }
}
```
